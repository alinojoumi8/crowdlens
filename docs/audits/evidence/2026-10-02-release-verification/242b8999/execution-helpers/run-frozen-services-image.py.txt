def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def save():(root/'services-image-receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
def run(label,cmd,check=True,env=baseenv,cwd=source,timeout=180):
    log=logs/(label+'.log');start=time.monotonic()
    with log.open('w') as stream:
        p=subprocess.Popen(cmd,cwd=cwd,env=env,stdout=stream,stderr=subprocess.STDOUT,start_new_session=True)
        try:code=p.wait(timeout=timeout)
        except subprocess.TimeoutExpired:
            os.killpg(p.pid,signal.SIGTERM)
            try:p.wait(timeout=5)
            except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.wait()
            code=124
    receipt['commands'].append({'label':label,'argv':cmd,'cwd':str(cwd),'exit_code':code,'seconds':round(time.monotonic()-start,3),'log':str(log),'sha256':sha(log)});save()
    if check and code:raise RuntimeError(label+' failed: '+str(code))
    return log.read_text()
def http(label,url):
    with urllib.request.build_opener(urllib.request.ProxyHandler({})).open(url,timeout=10) as response:
        content=response.read();status=response.status
    p=logs/(label+'.response');p.write_bytes(content)
    receipt['checks'][label]={'url':url,'status':status,'path':str(p),'sha256':sha(p)};save()
    assert status==200
    try:return json.loads(content)
    except ValueError:return content.decode()
